cmake_minimum_required(VERSION 3.20)
project(aethron_raster_native LANGUAGES CXX)
include(CTest)
option(AETHRON_SANITIZERS "Enable ASan and UBSan for kernel checks" OFF)
add_executable(kernel_test tests/kernel_test.cpp)
target_compile_features(kernel_test PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(kernel_test PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(kernel_test PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  if(AETHRON_SANITIZERS)
    target_compile_options(kernel_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(kernel_test PRIVATE -fsanitize=address,undefined)
  endif()
endif()
add_test(NAME bounded_kernel COMMAND kernel_test)
